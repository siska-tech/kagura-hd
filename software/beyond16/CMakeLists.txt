cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(beyond16 C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
set(MT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../midnight_terminal")
set(MT_PIOASM "${MT_ROOT}/out/pioasm/pioasm.exe" CACHE FILEPATH "Existing host pioasm")
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/kagura_host.pio.h
 COMMAND ${MT_PIOASM} -o c-sdk ${MT_ROOT}/vendor/kagura_host.pio ${CMAKE_CURRENT_BINARY_DIR}/kagura_host.pio.h
 DEPENDS ${MT_ROOT}/vendor/kagura_host.pio VERBATIM)
add_executable(beyond16 src/b16_profile.c src/b16.c src/rp2040_main.c src/b16_qspi.c ${MT_ROOT}/src/mt_lz.c ../kagura_fb/kagura_fb.c ${CMAKE_CURRENT_BINARY_DIR}/kagura_host.pio.h)
target_include_directories(beyond16 PRIVATE src ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(beyond16 pico_stdlib hardware_pio hardware_dma)
target_link_options(beyond16 PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/beyond16.elf.map")
target_compile_options(beyond16 PRIVATE -O2 -Wall -Wextra -fstack-usage)
option(B16_HOST_FAST "Constant clock waits and bounded FIFO deadline checks" ON)
target_compile_definitions(beyond16 PRIVATE B16_HOST_FAST=$<BOOL:${B16_HOST_FAST}>)
target_compile_definitions(beyond16 PRIVATE PICO_STACK_SIZE=0x2000 KFB_PACK_DESCRIPTOR=1)
pico_enable_stdio_uart(beyond16 0)
pico_enable_stdio_usb(beyond16 1)
add_custom_command(TARGET beyond16 POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:beyond16> ${CMAKE_CURRENT_BINARY_DIR}/beyond16.bin)
# No upload, flash, UF2, SRAM load or device discovery target.

set(B16_PROFILE 0 CACHE STRING "0 OFF, 1 SUMMARY, 2 DETAILED")
set(B16_TRANSPORT 0 CACHE STRING "0 CPU, 1 DMA writes >=64, 2 Hybrid measured threshold")
set(B16_DMA_THRESHOLD 0 CACHE STRING "Hybrid measured threshold; 0 disables DMA")
set(B16_SCHEDULER 0 CACHE STRING "0 relative33, 1 absolute30Hz")
set(B16_WORK_ROUNDS 0 CACHE STRING "Synthetic work rounds per bounded batch (0 disabled)")
target_compile_definitions(beyond16 PRIVATE B16_PROFILE=${B16_PROFILE} B16_TRANSPORT=${B16_TRANSPORT} B16_DMA_THRESHOLD=${B16_DMA_THRESHOLD} B16_SCHEDULER=${B16_SCHEDULER} B16_WORK_ROUNDS=${B16_WORK_ROUNDS} B16_KFB_PROFILE=1)
target_include_directories(beyond16 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

option(B16_COOP "Experimental one-entry CPU seed precompute during non-PRESENT DONE polling" OFF)
if(B16_COOP AND NOT B16_TRANSPORT STREQUAL "0")
 message(FATAL_ERROR "Evaluate cooperative GPU waiting separately from DMA")
endif()
target_compile_definitions(beyond16 PRIVATE B16_COOP=$<BOOL:${B16_COOP}>)
set(B16_GPU_QUEUE 0 CACHE STRING "Experimental approved GPU queue: 0 serial, 8 queued FILL/PALETTE")
if(NOT B16_GPU_QUEUE MATCHES "^(0|8)$")
 message(FATAL_ERROR "B16_GPU_QUEUE must be 0 or 8")
endif()
target_compile_definitions(beyond16 PRIVATE B16_GPU_QUEUE=${B16_GPU_QUEUE})
option(B16_QUEUE_METRICS "Sample queue retirement and barrier timings 1/64" OFF)
option(B16_MERGE_LETTERING "Pixel-equivalent vertical merging of finale glyph spans" OFF)
target_compile_definitions(beyond16 PRIVATE B16_QUEUE_METRICS=$<BOOL:${B16_QUEUE_METRICS}> B16_MERGE_LETTERING=$<BOOL:${B16_MERGE_LETTERING}>)
option(B16_COMPLETION_BURST "Experimental first status/tag/error snapshot; later polls remain status-only" OFF)
target_compile_definitions(beyond16 PRIVATE B16_COMPLETION_BURST=$<BOOL:${B16_COMPLETION_BURST}>)
